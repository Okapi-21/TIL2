## 新規証明書更新

------

本番環境にて以下のコマンドを実行

```bash
chmod 700 renew_letsencrypt.sh
sudo ./renew_letsencrypt.sh

#証明書更新が行われるかどうかテストを行いたい場合（このまま実行してください、パスを変えずに）
sudo /usr/local/sbin/renew_letsencrypt.sh --dry-run
```

以下のシェルスクリプトを実行する

```bash
#!/bin/bash
# =============================================================================
# Let's Encrypt 証明書 自動更新スクリプト
# 方式    : certbot --apache（Apache稼働中に更新）
# 再起動  : 証明書が実際に更新された場合のみ graceful reload
# 実行    : cron による定期実行 または 手動実行
#
# 使い方:
#   本番実行（実際に証明書を更新）  : ./renew_letsencrypt.sh
#   テスト実行（模擬実行のみ）      : ./renew_letsencrypt.sh --dry-run
# =============================================================================

# --- 定数 -------------------------------------------------------------------
LOG_FILE="/var/log/letsencrypt/renew_letsencrypt.log"
CERTBOT="/usr/bin/certbot"

# --- オプション解析 ---------------------------------------------------------
DRY_RUN=false
for ARG in "$@"; do
    case "$ARG" in
        --dry-run)
            DRY_RUN=true
            ;;
        *)
            echo "不明なオプション: $ARG"
            echo "使い方: $0 [--dry-run]"
            exit 1
            ;;
    esac
done

# --- 関数 -------------------------------------------------------------------
log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a "$LOG_FILE"
}

# --- 前提チェック -----------------------------------------------------------
if [ "$(id -u)" -ne 0 ]; then
    log "[ERROR] このスクリプトは root 権限で実行してください。"
    exit 1
fi

if [ ! -x "$CERTBOT" ]; then
    log "[ERROR] certbot が見つかりません: $CERTBOT"
    exit 1
fi

mkdir -p "$(dirname "$LOG_FILE")"

# --- 実行モードの表示 -------------------------------------------------------
if "$DRY_RUN"; then
    log "===== 証明書更新 開始 【テスト実行（--dry-run）】====="
    log "[INFO] 模擬実行モードです。証明書ファイルの書き換えは行いません。"
    log "[INFO] Let's Encrypt テスト用サーバーに接続して動作確認のみ行います。"
else
    log "===== 証明書更新 開始 【本番実行】====="
fi

# --- 証明書更新 -------------------------------------------------------------

# --dry-run の場合はオプションを追加、Apache reload もスキップ
if "$DRY_RUN"; then
    $CERTBOT renew \
        --apache \
        --dry-run \
        >> "$LOG_FILE" 2>&1
else
    $CERTBOT renew \
        --apache \
        --deploy-hook "systemctl reload httpd && echo '[$(date '+%Y-%m-%d %H:%M:%S')] [INFO] Apache reload 完了（証明書更新検知）' >> $LOG_FILE" \
        >> "$LOG_FILE" 2>&1
fi

EXIT_CODE=$?

# --- 結果判定 ---------------------------------------------------------------
if [ $EXIT_CODE -eq 0 ]; then
    if "$DRY_RUN"; then
        log "[INFO] テスト実行 正常終了 → 本番実行でも正しく動作する見込みです。"
    else
        log "[INFO] 証明書更新 正常終了（更新不要または更新成功）"
    fi
else
    if "$DRY_RUN"; then
        log "[ERROR] テスト実行 異常終了（終了コード: $EXIT_CODE）→ 本番実行前に問題を解消してください。"
    else
        log "[ERROR] 証明書更新 異常終了（終了コード: $EXIT_CODE）"
    fi
    exit $EXIT_CODE
fi

log "===== 証明書更新処理 終了 ====="
exit 0
```
