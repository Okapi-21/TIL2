## 新規証明書取得

------

本番環境にて、以下のコマンドを実行

```bash
#配置・権限付与
chmod 700 setup_letsencrypt.sh
sudo ./setup_letsencrypt.sh

#実行後は構文確認して再起動
apachectl configtest
systemctl restart httpd 
```



以下のシェルスクリプトの実行を行う

実行前に、スクリプト冒頭の２行の編集を忘れずに

```bash
#!/bin/bash
# =============================================================================
# Let's Encrypt 新規登録・ZeroSSLからの移行スクリプト（acme.sh版）
# 対象    : 単一ドメイン（wwwなし）
# 方式    : acme.sh + Apacheプラグイン（Apache稼働中に登録）
# 用途    : 初回登録 + cron による自動更新設定まで一括実施
# 前提    : curl がインストール済みであること
# =============================================================================

# --- 設定（実行前に必ず変更してください） ------------------------------------
DOMAIN="example.com"           # 対象ドメイン名に変更してください
EMAIL="admin@example.com"      # 証明書期限通知を受け取るメールアドレスに変更してください

# --- 定数 -------------------------------------------------------------------
LOG_FILE="/var/log/letsencrypt/setup_acme.log"
ACME_INSTALL_DIR="/root/.acme.sh"
ACME_SH="${ACME_INSTALL_DIR}/acme.sh"
CERT_DIR="/etc/letsencrypt/live/${DOMAIN}"
RENEW_SCRIPT="/usr/local/sbin/renew_acme.sh"
CRON_FILE="/etc/cron.d/acme_renew"

# --- 関数 -------------------------------------------------------------------
log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a "$LOG_FILE"
}

error_exit() {
    log "[ERROR] $1"
    exit 1
}

# --- 前提チェック -----------------------------------------------------------
if [ "$(id -u)" -ne 0 ]; then
    error_exit "このスクリプトは root 権限で実行してください。"
fi

if [ "$DOMAIN" = "example.com" ]; then
    error_exit "DOMAIN を実際のドメイン名に変更してから実行してください。"
fi

if [ "$EMAIL" = "admin@example.com" ]; then
    error_exit "EMAIL を実際のメールアドレスに変更してから実行してください。"
fi

if ! command -v curl > /dev/null 2>&1; then
    error_exit "curl が見つかりません。curl をインストールしてから実行してください。"
fi

mkdir -p "$(dirname "$LOG_FILE")"
mkdir -p "$CERT_DIR"

# =============================================================================
# STEP 1: acme.sh のダウンロードとインストール
# =============================================================================
log "===== STEP 1: acme.sh インストール ====="

if [ -f "$ACME_SH" ]; then
    log "[INFO] acme.sh はすでにインストール済みです。スキップします。"
else
    log "[INFO] acme.sh をダウンロードしています..."

    curl -s https://get.acme.sh | sh -s email="$EMAIL" >> "$LOG_FILE" 2>&1

    if [ $? -ne 0 ]; then
        error_exit "acme.sh のインストールに失敗しました。$LOG_FILE を確認してください。"
    fi

    log "[INFO] acme.sh インストール完了"
fi

# acme.sh のパスを再確認
if [ ! -f "$ACME_SH" ]; then
    error_exit "acme.sh が見つかりません: $ACME_SH"
fi

# =============================================================================
# STEP 2: Let's Encrypt を認証局として設定
# =============================================================================
log "===== STEP 2: 認証局を Let's Encrypt に設定 ====="

# acme.sh のデフォルト認証局は ZeroSSL のため、明示的に Let's Encrypt を指定
"$ACME_SH" --set-default-ca --server letsencrypt >> "$LOG_FILE" 2>&1

if [ $? -ne 0 ]; then
    error_exit "認証局の設定に失敗しました。$LOG_FILE を確認してください。"
fi

log "[INFO] 認証局を Let's Encrypt に設定完了"

# =============================================================================
# STEP 3: 証明書の新規取得
# =============================================================================
log "===== STEP 3: Let's Encrypt 証明書 新規取得 開始 ====="
log "対象ドメイン : $DOMAIN"
log "通知メール   : $EMAIL"

# --webroot を使い Apache 稼働中に取得
# Apache の公開ディレクトリ（/var/www/html）経由でドメイン所有確認を行う
"$ACME_SH" --issue \
    --webroot /var/www/html \
    --domain "$DOMAIN" \
    --server letsencrypt \
    >> "$LOG_FILE" 2>&1

ISSUE_EXIT=$?

# 終了コード 2 は「証明書がすでに有効で更新不要」を意味する（正常扱い）
if [ $ISSUE_EXIT -ne 0 ] && [ $ISSUE_EXIT -ne 2 ]; then
    error_exit "証明書の取得に失敗しました（終了コード: $ISSUE_EXIT）。$LOG_FILE を確認してください。"
fi

log "[INFO] 証明書取得 完了"

# =============================================================================
# STEP 4: 証明書を Apache 用のディレクトリにインストール
# =============================================================================
log "===== STEP 4: 証明書を Apache 用ディレクトリにインストール ====="

"$ACME_SH" --install-cert \
    --domain "$DOMAIN" \
    --cert-file      "${CERT_DIR}/cert.pem" \
    --key-file       "${CERT_DIR}/privkey.pem" \
    --fullchain-file "${CERT_DIR}/fullchain.pem" \
    --ca-file        "${CERT_DIR}/chain.pem" \
    --reloadcmd      "systemctl reload httpd" \
    >> "$LOG_FILE" 2>&1

if [ $? -ne 0 ]; then
    error_exit "証明書のインストールに失敗しました。$LOG_FILE を確認してください。"
fi

log "[INFO] 証明書インストール完了"
log "  cert.pem      : ${CERT_DIR}/cert.pem"
log "  privkey.pem   : ${CERT_DIR}/privkey.pem"
log "  fullchain.pem : ${CERT_DIR}/fullchain.pem"
log "  chain.pem     : ${CERT_DIR}/chain.pem"

# =============================================================================
# STEP 5: Apache の SSL 設定切り替え案内
# =============================================================================
log "===== STEP 5: Apache SSL 設定の切り替え ====="
log ""
log "[NOTICE] Apache の SSL 設定ファイルを以下のパスに手動で書き換えてください："
log "         SSLCertificateFile    ${CERT_DIR}/fullchain.pem"
log "         SSLCertificateKeyFile ${CERT_DIR}/privkey.pem"
log "         SSLCACertificateFile  ${CERT_DIR}/chain.pem"
log ""
log "  設定変更後、構文チェックと再起動を行ってください："
log "  apachectl configtest"
log "  systemctl reload httpd"

# =============================================================================
# STEP 6: 自動更新スクリプトの設置
# =============================================================================
log "===== STEP 6: 自動更新スクリプト設置 ====="

cat > "$RENEW_SCRIPT" << 'EOF'
#!/bin/bash
# =============================================================================
# Let's Encrypt 証明書 自動更新スクリプト（acme.sh版）
# 方式    : acme.sh（Apache稼働中に更新）
# 再起動  : 証明書が実際に更新された場合のみ graceful reload
# 実行    : cron による定期実行 または 手動実行
#
# 使い方:
#   本番実行（実際に証明書を更新）  : ./renew_acme.sh
#   テスト実行（模擬実行のみ）      : ./renew_acme.sh --dry-run
# =============================================================================

LOG_FILE="/var/log/letsencrypt/renew_acme.log"
ACME_SH="/root/.acme.sh/acme.sh"

# --- オプション解析 ---------------------------------------------------------
DRY_RUN=false
for ARG in "$@"; do
    case "$ARG" in
        --dry-run)
            DRY_RUN=true
            ;;
        *)
            echo "不明なオプション: $ARG"
            echo "使い方: $0 [--dry-run]"
            exit 1
            ;;
    esac
done

# --- 関数 -------------------------------------------------------------------
log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a "$LOG_FILE"
}

# --- 前提チェック -----------------------------------------------------------
if [ "$(id -u)" -ne 0 ]; then
    log "[ERROR] このスクリプトは root 権限で実行してください。"
    exit 1
fi

if [ ! -f "$ACME_SH" ]; then
    log "[ERROR] acme.sh が見つかりません: $ACME_SH"
    exit 1
fi

mkdir -p "$(dirname "$LOG_FILE")"

# --- 実行モードの表示 -------------------------------------------------------
if "$DRY_RUN"; then
    log "===== 証明書更新 開始 【テスト実行（--dry-run）】====="
    log "[INFO] 模擬実行モードです。証明書ファイルの書き換えは行いません。"
    log "[INFO] Let's Encrypt テスト用サーバーに接続して動作確認のみ行います。"
else
    log "===== 証明書更新 開始 【本番実行】====="
fi

# --- 証明書更新 -------------------------------------------------------------
if "$DRY_RUN"; then
    # --dry-run の場合はステージング環境（テスト用サーバー）で模擬実行
    "$ACME_SH" --renew-all \
        --server letsencrypt_test \
        >> "$LOG_FILE" 2>&1
else
    # 本番実行（--reloadcmd は install-cert 時に登録済みのため指定不要）
    "$ACME_SH" --renew-all \
        --server letsencrypt \
        >> "$LOG_FILE" 2>&1
fi

EXIT_CODE=$?

# --- 結果判定 ---------------------------------------------------------------
# 終了コード 2 は「更新不要（有効期限が十分残っている）」を意味する（正常扱い）
if [ $EXIT_CODE -eq 0 ] || [ $EXIT_CODE -eq 2 ]; then
    if "$DRY_RUN"; then
        log "[INFO] テスト実行 正常終了 → 本番実行でも正しく動作する見込みです。"
    else
        log "[INFO] 証明書更新 正常終了（更新不要または更新成功）"
    fi
else
    if "$DRY_RUN"; then
        log "[ERROR] テスト実行 異常終了（終了コード: $EXIT_CODE）→ 本番実行前に問題を解消してください。"
    else
        log "[ERROR] 証明書更新 異常終了（終了コード: $EXIT_CODE）"
    fi
    exit $EXIT_CODE
fi

log "===== 証明書更新処理 終了 ====="
exit 0
EOF

chmod 700 "$RENEW_SCRIPT"
log "[INFO] 自動更新スクリプト設置完了 : $RENEW_SCRIPT"

# =============================================================================
# STEP 7: cron 登録（毎日 午前3時に実行）
# =============================================================================
log "===== STEP 7: cron 登録 ====="

cat > "$CRON_FILE" << EOF
# Let's Encrypt 証明書自動更新（毎日 03:00 実行）
0 3 * * * root $RENEW_SCRIPT
EOF

chmod 644 "$CRON_FILE"
log "[INFO] cron 登録完了 : $CRON_FILE"

# =============================================================================
# 完了サマリー
# =============================================================================
log "===== セットアップ完了 ====="
log ""
log "【次にやること】"
log "  1. Apache の SSL 設定ファイルを以下のパスに書き換える"
log "     SSLCertificateFile    ${CERT_DIR}/fullchain.pem"
log "     SSLCertificateKeyFile ${CERT_DIR}/privkey.pem"
log "     SSLCACertificateFile  ${CERT_DIR}/chain.pem"
log ""
log "  2. 設定に問題ないか構文チェックする"
log "     apachectl configtest"
log ""
log "  3. Apache を再起動して証明書を反映する"
log "     systemctl reload httpd"
log ""
log "  4. ブラウザで https://$DOMAIN にアクセスし"
log "     Let's Encrypt 証明書が表示されることを確認する"
log ""
log "  5. 動作確認（dry-run）"
log "     $RENEW_SCRIPT --dry-run"
log ""
log "【自動更新】"
log "  cron 設定 : $CRON_FILE"
log "  更新ログ  : /var/log/letsencrypt/renew_acme.log"

exit 0
```
